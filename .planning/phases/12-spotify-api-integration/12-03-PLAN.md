# Plan 12-03: Playlist Fetch & Display Endpoints

**Status:** Pending  
**Phase:** 12 — Spotify API Integration  
**Milestone:** v1.5

## Objective

Create REST API endpoints for fetching and displaying user's Spotify playlists in Harmonix.

## Tasks

### 1. Create Routes File (`server/routes/spotify.js`)

Implement the following endpoints:

#### GET `/api/spotify/auth/url`
- Generate OAuth state parameter (CSRF protection)
- Generate PKCE code challenge
- Store state and verifier in session/temp storage
- Return authorization URL to frontend

**Response:**
```json
{
  "authUrl": "https://accounts.spotify.com/authorize?..."
}
```

#### GET `/api/spotify/callback`
- Handle redirect from Spotify
- Extract `code` and `state` from query params
- Validate state parameter
- Exchange code for tokens using `spotifyService`
- Store tokens in database
- Redirect to frontend success page or return tokens

**Query Parameters:**
- `code` — Authorization code from Spotify
- `state` — State parameter for CSRF validation

**Response:**
- Redirect to frontend with success indicator, OR
- JSON response with connection status

#### GET `/api/spotify/status`
- Check if user has connected Spotify account
- Return connection status and basic profile info

**Response:**
```json
{
  "connected": true,
  "spotifyUserId": "1234567890",
  "displayName": "John Doe",
  "scopes": ["playlist-read-private", "playlist-modify-public"]
}
```

#### DELETE `/api/spotify/disconnect`
- Remove stored tokens from database
- Clear any cached data
- Return confirmation

**Response:**
```json
{
  "message": "Spotify disconnected successfully"
}
```

#### GET `/api/spotify/playlists`
- Fetch user's Spotify playlists
- Support pagination with `limit` and `offset` query params
- Cache results temporarily (optional)

**Query Parameters:**
- `limit` — Number of playlists (default: 20, max: 50)
- `offset` — Pagination offset (default: 0)

**Response:**
```json
{
  "playlists": [
    {
      "id": "37i9dQZF1DXcBWIGoYBM5M",
      "name": "Today's Top Hits",
      "description": "The hottest tracks right now",
      "owner": "Spotify",
      "tracks_total": 50,
      "image_url": "https://...",
      "external_url": "https://open.spotify.com/playlist/..."
    }
  ],
  "total": 150,
  "limit": 20,
  "offset": 0
}
```

#### GET `/api/spotify/playlists/:playlistId/tracks`
- Fetch tracks from a specific Spotify playlist
- Return track details including Spotify URI

**Response:**
```json
{
  "playlist": {
    "id": "...",
    "name": "...",
    "owner": "..."
  },
  "tracks": [
    {
      "id": "3xKsf9qdS1CyvXSMEid6g8",
      "name": "Song Title",
      "artist": "Artist Name",
      "album": "Album Name",
      "uri": "spotify:track:3xKsf9qdS1CyvXSMEid6g8",
      "duration_ms": 180000,
      "added_at": "2024-01-01T00:00:00Z"
    }
  ],
  "total": 50
}
```

### 2. Apply Rate Limiting

- Apply `spotifyRateLimiter` to all `/api/spotify/*` routes
- Return 429 status when limit exceeded

### 3. Error Handling

Standardize error responses:

```javascript
// Not connected
res.status(400).json({ error: 'Spotify account not connected' });

// Token expired and refresh failed
res.status(401).json({ error: 'Spotify authentication expired, please reconnect' });

// API error
res.status(502).json({ error: 'Failed to fetch from Spotify' });

// Rate limited
res.status(429).json({ error: 'Too many requests to Spotify' });
```

## Example Route Structure

```javascript
// server/routes/spotify.js
const express = require('express');
const router = express.Router();
const spotifyService = require('../services/spotifyService');
const { authenticateToken } = require('../auth');
const spotifyRateLimiter = require('../middleware/spotifyRateLimiter');

// Apply rate limiter to all routes
router.use(spotifyRateLimiter);

// Get OAuth URL (requires auth to associate with user)
router.get('/auth/url', authenticateToken, (req, res) => {
  try {
    const { authUrl, state } = spotifyService.getAuthUrl(req.user.id);
    // Store state in temp storage
    res.json({ authUrl });
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

// OAuth callback
router.get('/callback', async (req, res) => {
  try {
    const { code, state } = req.query;
    // Validate state, exchange code, save tokens
    await spotifyService.handleCallback(code, state);
    res.redirect('/settings?spotify=connected');
  } catch (err) {
    res.redirect('/settings?spotify=error');
  }
});

// Connection status
router.get('/status', authenticateToken, (req, res) => {
  try {
    const status = await spotifyService.getConnectionStatus(req.user.id);
    res.json(status);
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

// Disconnect
router.delete('/disconnect', authenticateToken, async (req, res) => {
  try {
    await spotifyService.deleteTokens(req.user.id);
    res.json({ message: 'Disconnected successfully' });
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

// Get playlists
router.get('/playlists', authenticateToken, async (req, res) => {
  try {
    const { limit = 20, offset = 0 } = req.query;
    const playlists = await spotifyService.getUserPlaylists(
      req.user.id,
      parseInt(limit),
      parseInt(offset)
    );
    res.json(playlists);
  } catch (err) {
    handleSpotifyError(res, err);
  }
});

// Get playlist tracks
router.get('/playlists/:id/tracks', authenticateToken, async (req, res) => {
  try {
    const tracks = await spotifyService.getPlaylistTracks(
      req.user.id,
      req.params.id
    );
    res.json(tracks);
  } catch (err) {
    handleSpotifyError(res, err);
  }
});

module.exports = router;
```
