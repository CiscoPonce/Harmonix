### 2. Export Endpoints

#### POST `/api/spotify/export`

Export a Harmonix playlist to Spotify.

**Request Body:**
```json
{
  "playlistId": "harmonix_playlist_id",
  "targetName": "My Harmonix Playlist", // Optional, defaults to Harmonix playlist name
  "isPublic": false
}
```

**Process:**
1. Fetch Harmonix playlist with songs
2. Match each song to Spotify (using cache + API)
3. Create new Spotify playlist
4. Add matched tracks to playlist
5. Update `song_cache` with new Spotify URIs
6. Return export result

**Response:**
```json
{
  "success": true,
  "spotifyPlaylist": {
    "id": "spotify_playlist_id",
    "name": "My Harmonix Playlist",
    "url": "https://open.spotify.com/playlist/...",
    "uri": "spotify:playlist:..."
  },
  "matchReport": {
    "total": 20,
    "matched": 18,
    "unmatched": 2,
    "matches": [
      {
        "harmonixSongId": "...",
        "title": "Song Title",
        "artist": "Artist",
        "spotifyUri": "spotify:track:...",
        "confidence": "high"
      }
    ],
    "unmatchedTracks": [
      {
        "harmonixSongId": "...",
        "title": "Unknown Song",
        "artist": "Unknown Artist",
        "reason": "no_match_found"
      }
    ]
  }
}
```

### 3. Bulk Matching Optimization

To avoid rate limits during large exports:

- **Batch requests:** Group 5-10 search requests together
- **Use cache first:** Always check `song_cache.spotify_uri` before API call
- **Parallel processing:** Use `Promise.all()` with concurrency limit (max 5 concurrent)
- **Exponential backoff:** On 429, wait 2^n seconds before retry
- **Progress tracking:** For very large playlists, return progress endpoint

## Example Export Flow

```javascript
// POST /api/spotify/export
async function exportPlaylist(userId, playlistId, options) {
  // 1. Get user's Spotify token
  const token = await spotifyService.getToken(userId);
  
  // 2. Fetch Harmonix playlist
  const harmonixPlaylist = db.prepare(`
    SELECT p.*, ps.song_id, sc.track_json
    FROM playlists p
    JOIN playlist_songs ps ON ps.playlist_id = p.id
    LEFT JOIN song_cache sc ON sc.song_id = ps.song_id
    WHERE p.id = ? AND p.user_id = ?
  `).all(playlistId, userId);
  
  // 3. Match songs
  const matches = [];
  const unmatched = [];
  
  for (const song of harmonixPlaylist.songs) {
    let spotifyUri = song.track_json?.spotify_uri;
    
    if (!spotifyUri) {
      // Search Spotify
      const track = await spotifyService.searchTrack(
        song.track_json.artist,
        song.track_json.title,
        token.access_token
      );
      
      if (track) {
        spotifyUri = track.uri;
        // Cache the result
        await spotifyService.cacheSpotifyUri(song.song_id, spotifyUri);
      }
    }
    
    if (spotifyUri) {
      matches.push({ ...song, spotifyUri });
    } else {
      unmatched.push(song);
    }
  }
  
  // 4. Create Spotify playlist
  const spotifyPlaylist = await spotifyService.createPlaylist(
    token.access_token,
    userId,
    options.targetName || harmonixPlaylist.name,
    `Exported from Harmonix on ${new Date().toISOString()}`,
    options.isPublic || false
  );
  
  // 5. Add tracks
  if (matches.length > 0) {
    await spotifyService.addTracksToPlaylist(
      token.access_token,
      spotifyPlaylist.id,
      matches.map(m => m.spotifyUri)
    );
  }
  
  // 6. Return result
  return {
    success: true,
    spotifyPlaylist: {
      id: spotifyPlaylist.id,
      name: spotifyPlaylist.name,
      url: spotifyPlaylist.external_urls.spotify,
      uri: spotifyPlaylist.uri
    },
    matchReport: {
      total: harmonixPlaylist.songs.length,
      matched: matches.length,
      unmatched: unmatched.length,
      matches: matches.map(m => ({
        title: m.track_json.title,
        artist: m.track_json.artist,
        spotifyUri: m.spotifyUri,
        confidence: 'high'
      })),
      unmatchedTracks: unmatched.map(u => ({
        title: u.track_json.title,
        artist: u.track_json.artist,
        reason: 'no_match_found'
      }))
    }
  };
}
```
