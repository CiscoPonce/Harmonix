<task type="auto">
  <name>Task 1: Implement Deezer Search & Track Proxy</name>
  <files>server/index.js</files>
  <action>
    Add `GET /api/search` and `GET /api/tracks/:id` endpoints to `server/index.js`.
    - `/api/search?q=...`: Proxies `https://api.deezer.com/search?q={query}`.
    - `/api/tracks/:id`: Proxies `https://api.deezer.com/track/{id}`.
    - Calculate `preview_offset` in the `:id` endpoint using the research heuristic:
      - If `duration > 60`: offset = 30
      - If `30 < duration <= 60`: offset = duration - 30
      - If `duration <= 30`: offset = 0
    - Return a cleaned object with `id`, `title`, `artist`, `preview`, `duration`, and `preview_offset`.
  </action>
  <verify>
    <automated>curl -s "http://localhost:3001/api/tracks/132143" | grep -q "\"preview_offset\":30"</automated>
  </verify>
  <done>Endpoints return valid JSON with track metadata and correct offset calculations (per Checker feedback).</done>
</task>
