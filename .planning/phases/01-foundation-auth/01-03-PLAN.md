<task type="auto">
  <name>Task 1: Implement API Client & Auth Context</name>
  <files>client/lib/api.ts, client/context/AuthContext.tsx, client/hooks/useAuth.ts, client/.env.local</files>
  <action>
    - Create `client/.env.local` with `NEXT_PUBLIC_API_URL=http://localhost:3001/api`.
    - Create `client/lib/api.ts`:
      - Base fetch wrapper using `NEXT_PUBLIC_API_URL`.
      - Set `credentials: 'include'`.
      - Request interceptor logic: if response is 401, attempt POST `/api/auth/refresh`.
      - If refresh fails (401 from refresh endpoint), clear `accessToken` and redirect to `/login`.
    - Create `client/context/AuthContext.tsx`:
      - Manage `user` object and `accessToken` in state.
      - `login`, `register`, `logout` methods.
      - `useEffect` on mount to attempt initial token refresh.
    - Create `client/hooks/useAuth.ts` hook.
  </action>
  <verify>
    <automated>
      grep -q "NEXT_PUBLIC_API_URL" client/lib/api.ts && \
      grep -q "/api/auth/refresh" client/lib/api.ts && \
      grep -q "redirect" client/lib/api.ts || echo "Failure handling present"
    </automated>
  </verify>
  <done>Client-side auth state management and API client implemented with robust error handling.</done>
